Examples

public sealed class Account
{
    [Ignore]
    public string? Secret { get; set; }

    [Ignore(Condition = IgnoreCondition.WhenWritingNull)]
    public string? Comment { get; set; }
}

// Secret is never written; Comment is written only when it is non-null.
